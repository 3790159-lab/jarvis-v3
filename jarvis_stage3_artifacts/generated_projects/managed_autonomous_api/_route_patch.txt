@app.post("/tasks")
def create_task(req: TaskCreateRequest):
    task_id = f"task_{uuid.uuid4().hex[:8]}"

    # AUTO ROUTING
    if not req.assigned_agent:
        if req.task_type in ["echo", "write_file", "read_file", "list_dir", "run_python_tests"]:
            req.assigned_agent = "executor_agent"
        elif req.task_type == "run_shell_command":
            req.assigned_agent = "shell_agent"
        else:
            req.assigned_agent = "planner_agent"

    task = create_task_record(
        task_id=task_id,
        mission_id=req.mission_id,
        task_type=req.task_type,
        payload=req.payload,
        assigned_agent=req.assigned_agent,
    )

    append_runtime_log(f"TASK queued task_id={task_id} type={req.task_type} assigned_agent={req.assigned_agent}")

    return task
